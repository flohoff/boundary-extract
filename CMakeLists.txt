cmake_minimum_required(VERSION 3.16)
project(boundary-extract CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
find_package(BZip2 REQUIRED)
find_package(EXPAT REQUIRED)
find_library(GEOS_C geos_c REQUIRED)
add_executable(boundary-extract src/main.cpp)
target_link_libraries(boundary-extract Threads::Threads ZLIB::ZLIB BZip2::BZip2 EXPAT::EXPAT ${GEOS_C})
